  \pgfmathsetlength\pgf@x{#1}%

\let\lokadusepath\pgfusepath
\def\pgfusepath#1{%
  \iflokadprobes
    \global\advance\lokadpath by1
    \typeout{LOKAD-DASH \the\lokadpath|width=\number\pgflinewidth|phase=\lokadphase|pattern=\lokaddash|actions=#1|END}%
    \begingroup
      \pgfmathsetlength\pgf@x{\lokadphase}%
      \typeout{LOKAD-DASH-PHASE \the\lokadpath|\number\pgf@x|END}%
      \ifx\lokaddash\pgfutil@empty\else
        \pgfutil@for\lokaditem:=\lokaddash\do{\expandafter\lokadlength\expandafter{\lokaditem}}%
      \fi
    \endgroup
  \fi
  \lokadusepath{#1}%
}
